\begin{algorithm}[H]
\Input{Page identifier, Streamlit session state}
\Output{A page that either refreshes once or stays frozen until explicitly resumed}
Build a page-specific namespace from the current file name\;
Render sidebar controls for freeze mode and manual refresh\;
\If{the user clicks refresh}{
    Clear cached data and cached resources\;
    Allow exactly one full rerun\;
    Rerun the page\;
}
\If{freeze mode is enabled and the page already initialized and no one-time refresh is pending}{
    Show a freeze notice and stop page execution\;
}
Mark the page as initialized and reset the one-time refresh flag\;
\caption{Runtime Controls for Heavy Streamlit Pages}
\label{alg:runtime-controls}
\end{algorithm}
